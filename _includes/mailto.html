{%- comment -%} Builds a mailto: URL. Usage: href="{% include mailto.html type="health_check" %}" `type` is a key under `enquiries` in _data/profile.yml; omit it for a plain link. {%- endcomment -%} {%- assign mailto_enquiry = site.data.profile.enquiries[include.type] -%} mailto:{{ site.data.profile.email }} {%- if mailto_enquiry -%} ?subject={{ mailto_enquiry.subject | uri_escape | replace: "&", "%26" | replace: "?", "%3F" | replace: "#", "%23" }} {%- if mailto_enquiry.body -%} &body={{ mailto_enquiry.body | uri_escape | replace: "&", "%26" | replace: "?", "%3F" | replace: "#", "%23" }} {%- endif -%} {%- endif -%}
